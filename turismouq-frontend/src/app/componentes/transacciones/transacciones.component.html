Implementación en Paquete PL/SQL (Oracle 21c)

PROCEDURE sp_registrar_reserva_con_pago(
    p_id_cliente      IN  NUMBER,
    p_id_habitacion   IN  NUMBER,
    p_fecha_in        IN  DATE,
    p_fecha_out       IN  DATE,
    p_num_huespedes   IN  NUMBER,
    p_metodo_pago     IN  VARCHAR2,
    p_simular_fallo   IN  NUMBER,    -- 1: Simula rechazo bancario, 0: Normal
    p_id_reserva      OUT NUMBER,
    p_id_pago         OUT NUMBER,
    p_total           OUT NUMBER,
    p_estado_reserva  OUT VARCHAR2,
    p_mensaje         OUT VARCHAR2
) IS
    v_id_reserva NUMBER;
    v_total      NUMBER;
    v_metodo     VARCHAR2(30);
BEGIN
    -- 1. Crear la reserva (valida ORA-20001 fechas, ORA-20002 solapamiento, ORA-20003 aforo)
    sp_crear_reserva(p_id_cliente, p_id_habitacion, p_fecha_in, p_fecha_out, p_num_huespedes, v_id_reserva, v_total);
    p_id_reserva := v_id_reserva;
    p_total      := v_total;

    -- 2. Fijar SAVEPOINT antes del cobro financiero
    SAVEPOINT pto_reserva_creada;

    -- 3. Flujo de pago: Exitoso vs Rollback parcial simulado
    IF p_simular_fallo = 1 THEN
        ROLLBACK TO pto_reserva_creada;
        UPDATE RESERVA SET ESTADO = 'PENDIENTE' WHERE ID_RESERVA = v_id_reserva;
        COMMIT;
        p_id_pago := NULL;
        p_estado_reserva := 'PENDIENTE';
    ELSE
        -- Normalización de método de pago y cobro aprobado en PAGO
        INSERT INTO PAGO (fecha_pago, monto, metodo_pago, estado, id_reserva)
        VALUES (SYSDATE, v_total, v_metodo, 'APROBADO', v_id_reserva)
        RETURNING id_pago INTO p_id_pago;
        COMMIT;
        p_estado_reserva := 'CONFIRMADA';
    END IF;
END sp_registrar_reserva_con_pago;
